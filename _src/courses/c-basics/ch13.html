13.1需求：先写清楚要做什么

做一个命令行通讯录，支持：添加联系人、按姓名查找、按姓名删除、列出全部、保存到文件、从文件加载。数据放在内存里，程序退出前可以保存。

数据结构

动态数组，而不是链表

联系人数量不大，需要按姓名查找、遍历打印。动态数组内存连续、缓存友好、实现简单；链表在这里没有优势。

模块划分

接口与实现分离

contact.h 暴露类型和函数声明，contact.c 藏实现，main.c 只负责交互。

工程习惯

先定接口，再写实现。接口稳定了，main.c 和 contact.c 可以并行开发，测试也可以直接针对接口写，不必经过菜单。

13.2接口：contact.h

#ifndef CONTACT_H #define CONTACT_H #include #define NAME_MAX 24 #define PHONE_MAX 16 typedef struct { char name[NAME_MAX]; char phone[PHONE_MAX]; } Contact; typedef struct { Contact *items; size_t count; size_t cap; } Book; void book_init(Book *b); void book_free(Book *b); int book_add(Book *b, const char *name, const char *phone); int book_remove(Book *b, const char *name); Contact *book_find(Book *b, const char *name); void book_list(const Book *b); int book_save(const Book *b, const char *path); int book_load(Book *b, const char *path); #endif /* CONTACT_H */

约定：所有返回 int 的函数，0 表示成功，-1 表示失败；book_find 返回指针，找不到返回 NULL。这些约定要写在头文件注释里，调用者才不会猜。

13.3实现：contact.c

先补一课：文件读写

项目要把通讯录保存到 contacts.txt，用到 C 的文件 I/O。它只有四步：fopen 打开、fprintf / fgets 读写、fclose 关闭、检查每一步的返回值。

模式含义注意
"r"只读，文件必须存在不存在返回 NULL
"w"只写，创建或清空会丢掉原内容
"a"追加文件不存在则创建
"rb" / "wb"二进制读写Windows 上文本模式会转换换行符
#include int main(void) { FILE *fp = fopen("demo.txt", "w"); if (fp == NULL) { perror("fopen"); /* 打印失败原因 */ return 1; } if (fprintf(fp, "%s\t%d\n", "Ada", 42) < 0) { fclose(fp); return 1; } if (fclose(fp) != 0) { /* 缓冲区落盘也可能失败 */ return 1; } return 0; }

fgets 读到 EOF 或出错时返回 NULL，并且会把换行符一起读进来；fprintf 失败返回负数。每一个返回值都要检查，否则磁盘满、路径不存在时程序会“看起来正常但什么都没保存”。

#include "contact.h" #include #include #include void book_init(Book *b) { b->items = NULL; b->count = 0; b->cap = 0; } void book_free(Book *b) { free(b->items); book_init(b); } /* 容量不足时翻倍；返回 0 成功，-1 失败 */ static int book_reserve(Book *b, size_t need) { if (need <= b->cap) { return 0; } size_t cap = b->cap ? b->cap : 4; while (cap < need) { cap *= 2; } Contact *p = realloc(b->items, cap * sizeof *p); if (p == NULL) { return -1; } b->items = p; b->cap = cap; return 0; } int book_add(Book *b, const char *name, const char *phone) { if (name == NULL || phone == NULL) { return -1; } if (strlen(name) >= NAME_MAX || strlen(phone) >= PHONE_MAX) { return -1; /* 放不下，拒绝而不是截断 */ } if (book_find(b, name) != NULL) { return -1; /* 姓名重复 */ } if (book_reserve(b, b->count + 1) != 0) { return -1; } Contact *c = &b->items[b->count]; snprintf(c->name, sizeof c->name, "%s", name); snprintf(c->phone, sizeof c->phone, "%s", phone); b->count++; return 0; } Contact *book_find(Book *b, const char *name) { for (size_t i = 0; i < b->count; i++) { if (strcmp(b->items[i].name, name) == 0) { return &b->items[i]; } } return NULL; } int book_remove(Book *b, const char *name) { for (size_t i = 0; i < b->count; i++) { if (strcmp(b->items[i].name, name) == 0) { /* 用最后一个元素覆盖被删元素，避免搬移 */ b->items[i] = b->items[b->count - 1]; b->count--; return 0; } } return -1; } void book_list(const Book *b) { if (b->count == 0) { printf("（通讯录为空）\n"); return; } for (size_t i = 0; i < b->count; i++) { printf("%-16s %s\n", b->items[i].name, b->items[i].phone); } } int book_save(const Book *b, const char *path) { FILE *fp = fopen(path, "w"); if (fp == NULL) { return -1; } for (size_t i = 0; i < b->count; i++) { if (fprintf(fp, "%s\t%s\n", b->items[i].name, b->items[i].phone) < 0) { fclose(fp); return -1; } } return fclose(fp) == 0 ? 0 : -1; } int book_load(Book *b, const char *path) { FILE *fp = fopen(path, "r"); if (fp == NULL) { return -1; } char line[128]; while (fgets(line, sizeof line, fp) != NULL) { char *tab = strchr(line, '\t'); if (tab == NULL) { continue; /* 跳过格式不对的行 */ } *tab = '\0'; char *phone = tab + 1; size_t n = strlen(phone); if (n > 0 && phone[n - 1] == '\n') { phone[n - 1] = '\0'; } (void)book_add(b, line, phone); /* 重复或非法就跳过 */ } fclose(fp); return 0; }
标准怎么说

strlen(name) >= NAME_MAX 的判断必须在复制之前，因为 snprintf 虽然不会越界，但会静默截断。对“姓名”这种数据，截断比报错更糟——两个不同的人可能变成同一个名字。

13.5调试：从“跑不起来”到“知道为什么”

第一步：把警告当错误看

gcc -std=c11 -Wall -Wextra -Wpedantic -Werror -O2 main.c contact.c -o contacts

如果 -Werror 让编译失败，先逐条修掉警告，再谈逻辑。常见警告与含义：

警告通常意味着
-Wuninitialized变量没赋值就用了
-Wsign-compare有符号和无符号在比较，可能是逻辑错误
-Wformatprintf 的格式符和参数类型不匹配
-Wshadow内层变量遮蔽了外层同名变量
-Wconversion隐式窄化转换，可能丢数据（需手动开启）

第二步：用 GDB 看现场

gcc -std=c11 -g -O0 main.c contact.c -o contacts gdb ./contacts break book_add # 在函数入口下断点 run # 运行 next # 执行下一行（不进入函数） step # 进入函数 print b->count # 打印表达式 backtrace # 查看调用栈 watch b->count # 变量变化时停下来 continue # 继续 quit # 退出
工程习惯

崩溃时第一件事不是改代码，而是看 backtrace：确认崩在哪个函数、调用链是什么、参数是否合理。90% 的段错误来自空指针、越界和释放后使用，而这三类在栈里通常一眼能看出来。

第三步：边界测试清单

  • 空通讯录直接列表、查找、删除。
  • 添加第 1 个、第 5 个（触发扩容）、第 100 个联系人。
  • 姓名恰好 23 个字符（能放下）和 24 个字符（应拒绝）。
  • 重复姓名、空姓名、只有空格的姓名。
  • 保存后再加载，数据是否一致；加载不存在的文件是否崩溃。
  • 用 ASan 跑一遍全部操作。

项目练习 · 第 13 讲
  1. 给通讯录加“按电话前缀查找”功能，返回所有匹配项。
  2. 把删除改成“保持原顺序”（搬移后面的元素），比较两种实现的代价。
  3. 增加“修改电话”命令，并处理姓名不存在的情况。
  4. 为 book_add 写一组测试：正常、重名、姓名过长、电话过长、内存不足（可用宏模拟 malloc 失败）。
参考答案与思路

1. 新增 size_t book_find_prefix(const Book *b, const char *prefix, Contact *out, size_t max)，返回匹配个数，调用者提供输出数组，避免函数内部分配。

2. 保序删除用 memmove(&items[i], &items[i+1], (count-i-1)*sizeof *items)，时间复杂度 O(n)；覆盖法 O(1)，但顺序会变。数据量小时差别不大，接口语义要写清楚。

3. 先 book_find，为空返回错误；否则用 snprintf 写入并检查长度。

4. 测试要覆盖返回值、边界长度、重复姓名，以及分配失败时原数据是否保持不变（强异常安全）。

自测：项目里 book_remove 用“最后一个元素覆盖被删元素”，代价是什么？

覆盖法把最后一个元素搬到被删位置，时间 O(1)，但破坏了原有顺序。如果需要稳定顺序，就得用 memmove 搬移，代价是 O(n)。

本讲小结

  • 先定接口再写实现，模块边界清晰，测试才有抓手。
  • 动态数组的扩容、失败回滚、所有权释放要一次设计对。
  • 输入长度检查必须在复制之前，拒绝比静默截断安全。
  • 调试顺序：警告 → 复现 → 断点/栈 → 边界测试 → sanitizer。
  • 项目的价值不在代码量，而在“能解释每个设计取舍”。
