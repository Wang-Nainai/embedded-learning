10.1目标：一条能自恢复的接收链

把前面几讲拼起来：UART 中断收字节 → 环形缓冲区 → 帧解析状态机 → CRC 校验 → 命令处理 → 生成响应。要求：半包能续、粘包能拆、噪声能重同步、CRC 错能丢弃、长度非法能拒绝。

层职责本讲对应
字节接收中断读寄存器、入环形缓冲第 3 讲
分帧找帧头、读长度、收载荷、验 CRC第 8 讲
命令处理按 type 分发、校验参数、生成响应本讲
发送组帧、算 CRC、写 UART本讲
工程习惯

解析器不直接调用 printf 或硬件，只做“字节进、帧出”；命令处理也不直接写 UART，只生成响应缓冲区。这样整个协议栈可以在 PC 上跑单元测试，移植时只替换最外层的收发。

10.2接口：proto.h

#ifndef PROTO_H #define PROTO_H #include #include #define PROTO_SOF1 0xAAu #define PROTO_SOF2 0x55u #define PROTO_MAX_PAYLOAD 32u typedef enum { PROTO_OK = 0, PROTO_ERR_TYPE, PROTO_ERR_RANGE, PROTO_ERR_LEN, PROTO_ERR_CRC } proto_status_t; typedef enum { TYPE_GET_REQ = 0x01, TYPE_SET_REQ = 0x02, TYPE_GET_RSP = 0x81, TYPE_SET_RSP = 0x82, TYPE_ERROR = 0xFF } proto_type_t; typedef struct { uint8_t state; uint8_t type, seq, len, index; uint8_t payload[PROTO_MAX_PAYLOAD]; uint16_t crc_rx; } proto_parser_t; void proto_init(proto_parser_t *p); /* 喂一个字节；返回 1 表示收到完整帧（已通过 CRC） */ int proto_feed(proto_parser_t *p, uint8_t byte); /* 处理一帧，生成响应帧到 out；返回响应长度，0 表示不响应 */ size_t proto_handle(const proto_parser_t *p, uint8_t *out, size_t out_len); const char *proto_status_str(proto_status_t st); #endif

三个关键约定：

  • proto_feed 返回 1 表示“收到完整且 CRC 正确的帧”；返回负数表示“检测到错误并已重同步”。
  • 解析器只保存状态，不分配内存；payload 固定 32 字节。
  • proto_handle 只依赖解析结果，不依赖硬件。

10.3实现：proto.c

#include "proto.h" #include enum { S_SOF1 = 0, S_SOF2, S_TYPE, S_SEQ, S_LEN, S_PAYLOAD, S_CRC_LO, S_CRC_HI }; static uint16_t crc16(const uint8_t *d, size_t n) { uint16_t crc = 0xFFFFu; for (size_t i = 0; i < n; i++) { crc ^= d[i]; for (int b = 0; b < 8; b++) { crc = (crc & 1u) ? (uint16_t)((crc >> 1) ^ 0xA001u) : (uint16_t)(crc >> 1); } } return crc; } static uint16_t frame_crc(uint8_t type, uint8_t seq, uint8_t len, const uint8_t *payload) { uint8_t buf[PROTO_MAX_PAYLOAD + 3u]; buf[0] = type; buf[1] = seq; buf[2] = len; if (len > 0u) { memcpy(&buf[3], payload, len); } return crc16(buf, (size_t)len + 3u); } void proto_init(proto_parser_t *p) { memset(p, 0, sizeof *p); p->state = S_SOF1; } int proto_feed(proto_parser_t *p, uint8_t b) { switch (p->state) { case S_SOF1: if (b == PROTO_SOF1) p->state = S_SOF2; break; case S_SOF2: if (b == PROTO_SOF2) p->state = S_TYPE; else if (b != PROTO_SOF1) p->state = S_SOF1; break; case S_TYPE: p->type = b; p->state = S_SEQ; break; case S_SEQ: p->seq = b; p->state = S_LEN; break; case S_LEN: if (b > PROTO_MAX_PAYLOAD) { p->state = S_SOF1; /* 非法长度：重同步 */ return -1; } p->len = b; p->index = 0; p->state = (b == 0u) ? S_CRC_LO : S_PAYLOAD; break; case S_PAYLOAD: p->payload[p->index++] = b; if (p->index >= p->len) p->state = S_CRC_LO; break; case S_CRC_LO: p->crc_rx = b; p->state = S_CRC_HI; break; case S_CRC_HI: p->crc_rx |= (uint16_t)((uint16_t)b << 8); p->state = S_SOF1; if (frame_crc(p->type, p->seq, p->len, p->payload) == p->crc_rx) { return 1; } return -2; /* CRC 错 */ default: p->state = S_SOF1; break; } return 0; } static size_t build(uint8_t type, uint8_t seq, const uint8_t *payload, uint8_t len, uint8_t *out, size_t out_len) { size_t need = (size_t)len + 7u; /* 2 SOF + type + seq + len + payload + 2 CRC */ if (out_len < need) { return 0; } out[0] = PROTO_SOF1; out[1] = PROTO_SOF2; out[2] = type; out[3] = seq; out[4] = len; if (len > 0u) { memcpy(&out[5], payload, len); } uint16_t crc = frame_crc(type, seq, len, payload); out[5 + len] = (uint8_t)(crc & 0xFFu); out[6 + len] = (uint8_t)(crc >> 8); return need; } size_t proto_handle(const proto_parser_t *p, uint8_t *out, size_t out_len) { uint8_t rsp[PROTO_MAX_PAYLOAD]; switch (p->type) { case TYPE_GET_REQ: if (p->len != 1u || p->payload[0] > 3u) { rsp[0] = (uint8_t)PROTO_ERR_RANGE; return build(TYPE_ERROR, p->seq, rsp, 1u, out, out_len); } rsp[0] = p->payload[0]; rsp[1] = (uint8_t)(100u + p->payload[0]); return build(TYPE_GET_RSP, p->seq, rsp, 2u, out, out_len); case TYPE_SET_REQ: if (p->len != 2u || p->payload[0] > 3u) { rsp[0] = (uint8_t)PROTO_ERR_RANGE; return build(TYPE_ERROR, p->seq, rsp, 1u, out, out_len); } return build(TYPE_SET_RSP, p->seq, p->payload, 2u, out, out_len); default: rsp[0] = (uint8_t)PROTO_ERR_TYPE; return build(TYPE_ERROR, p->seq, rsp, 1u, out, out_len); } } const char *proto_status_str(proto_status_t st) { switch (st) { case PROTO_OK: return "ok"; case PROTO_ERR_TYPE: return "unknown type"; case PROTO_ERR_RANGE: return "out of range"; case PROTO_ERR_LEN: return "bad length"; case PROTO_ERR_CRC: return "crc error"; default: return "?"; } }
易错

frame_crc 里的临时缓冲区是 PROTO_MAX_PAYLOAD + 3，而 len 在进入状态机时已经校验过上限。如果哪天把上限改大却忘了改缓冲区，就会栈溢出——所以两处必须用同一个宏。

10.4主机测试：正常、分片、噪声、坏 CRC、坏长度

#include "proto.h" #include #include static size_t make(uint8_t type, uint8_t seq, const uint8_t *pl, uint8_t len, uint8_t *out) { size_t n = (size_t)len + 7u; out[0] = PROTO_SOF1; out[1] = PROTO_SOF2; out[2] = type; out[3] = seq; out[4] = len; if (len) memcpy(&out[5], pl, len); uint8_t buf[PROTO_MAX_PAYLOAD + 3u]; buf[0] = type; buf[1] = seq; buf[2] = len; if (len) memcpy(&buf[3], pl, len); uint16_t crc = 0xFFFFu; for (size_t i = 0; i < (size_t)len + 3u; i++) { crc ^= buf[i]; for (int b = 0; b < 8; b++) crc = (crc & 1u) ? (uint16_t)((crc >> 1) ^ 0xA001u) : (uint16_t)(crc >> 1); } out[5 + len] = (uint8_t)(crc & 0xFFu); out[6 + len] = (uint8_t)(crc >> 8); return n; } static void dump(const char *tag, const uint8_t *p, size_t n) { printf("%-22s", tag); for (size_t i = 0; i < n; i++) printf("%02X ", p[i]); printf("\n"); } int main(void) { proto_parser_t ps; proto_init(&ps); uint8_t frame[64], rsp[64]; uint8_t pl[4] = { 2, 55 }; size_t n = make(TYPE_GET_REQ, 7, pl, 1, frame); dump("GET ch2", frame, n); /* 逐字节喂入，模拟分片到达 */ for (size_t i = 0; i < n; i++) { int r = proto_feed(&ps, frame[i]); if (r == 1) { size_t rn = proto_handle(&ps, rsp, sizeof rsp); dump(" -> response", rsp, rn); } } /* 噪声 + 一帧写请求 */ uint8_t noise[] = { 0x00, 0xAA, 0x00, 0xAA, 0x00 }; for (size_t i = 0; i < sizeof noise; i++) (void)proto_feed(&ps, noise[i]); n = make(TYPE_SET_REQ, 8, pl, 2, frame); for (size_t i = 0; i < n; i++) { if (proto_feed(&ps, frame[i]) == 1) { size_t rn = proto_handle(&ps, rsp, sizeof rsp); dump("SET ch2=55", rsp, rn); } } /* 破坏 CRC */ n = make(TYPE_GET_REQ, 9, pl, 1, frame); frame[n - 1] ^= 0xFF; int last = 0; for (size_t i = 0; i < n; i++) last = proto_feed(&ps, frame[i]); printf("bad crc feed returned %d (%s)\n", last, last == -2 ? "crc error" : "other"); /* 非法长度：0x40 > 32 */ uint8_t bad_len[] = { PROTO_SOF1, PROTO_SOF2, 0x01, 0x0A, 0x40, 0x00 }; last = 0; for (size_t i = 0; i < sizeof bad_len; i++) { last = proto_feed(&ps, bad_len[i]); if (last < 0) break; } printf("bad length feed returned %d (%s)\n", last, last == -1 ? "resync" : "other"); return 0; } gcc -std=c11 -Wall -Wextra -Wpedantic -O2 proto.c main.c -o proto ./proto GET ch2 AA 55 01 07 01 02 30 48 -> response AA 55 81 07 02 02 66 39 E0 SET ch2=55 AA 55 82 08 02 02 37 BF 08 bad crc feed returned -2 (crc error) bad length feed returned -1 (resync)

这组输出是实际编译运行的结果。它验证了五件事：正常帧、逐字节分片、噪声后重同步、CRC 错误被拒绝、非法长度被拒绝。

10.5移植到 MCU

#include "proto.h" static proto_parser_t g_parser; void uart_rx_byte(uint8_t b) { int r = proto_feed(&g_parser, b); if (r == 1) { uint8_t rsp[64]; size_t n = proto_handle(&g_parser, rsp, sizeof rsp); if (n > 0) { uart_send(rsp, n); /* 发送可以阻塞，但不要在中断里做 */ } } else if (r < 0) { g_proto_errors++; } } /* 更好的做法：中断只入环形缓冲，主循环里 feed + handle */ void app_poll(void) { uint8_t b; while (ring_pop(&g_rx, &b)) { uart_rx_byte(b); } }
工程习惯

不要在 UART 中断里直接调用 uart_send 发响应：发送可能阻塞，且会让中断时间不可控。正确做法是中断只入队，主循环解析并发送。

10.6一致性测试清单

  • 正常请求 → 响应 type/seq 正确、CRC 正确
  • 把一帧拆成 1 字节、2 字节、随机长度喂入，结果一致
  • 两帧粘在一起，能连续解出两帧
  • 帧前插入随机噪声，能重新同步
  • CRC 错一字节，返回错误且不产生响应
  • 长度 = 0、长度 = 32、长度 = 33 的边界
  • 未知 type 返回错误帧，不崩溃
  • 参数越界返回错误码，不执行
  • 连续 10 万帧不泄漏、不越界（用 ASan 跑）
项目练习 · 第 10 讲
  1. 给协议加一个“版本”字段，放在 type 之后，并更新 CRC 覆盖范围。
  2. 增加 TYPE_HEARTBEAT（0x03），设备收到后回复当前 tick。
  3. 把固定 32 字节的 payload 改成“长度 + 指针”，讨论所有权与生命周期。
  4. 写一个模糊测试：随机生成字节流，跑 10 万次，确保不崩溃。
参考答案与思路

1. 在 frame_crc 与组帧函数里都加入版本字节；解析状态机加一个 S_VER 状态。

2. 新增 type，处理函数返回 4 字节 tick；注意请求/响应 type 要能区分。

3. 指针方案要明确“谁分配、谁释放、有效期到何时”；更安全的是调用者提供缓冲区（本讲做法）。

4. 用固定种子的伪随机数生成字节流，循环 feed；配合 ASan/UBSan，检查是否有越界或未定义行为。

自测：解析器收到 len = 0x40（上限 32）时，正确做法是？

非法长度说明当前帧已错位，继续读会把后续有效数据吞掉；截断则可能把噪声当载荷。立即重同步是唯一安全的选择。

本讲小结

  • 协议栈分层：接收、分帧、处理、发送，各自可测试。
  • 解析器只保存状态，不做硬件操作。
  • 长度上限与缓冲区大小必须用同一个宏。
  • 错误要能区分“CRC 错”和“长度非法”，并各自重同步。
  • 测试覆盖分片、粘包、噪声、边界与模糊输入。

  • 我能解释每个返回值的含义
