13.1需求：从字节流到一条命令

目标：做一个与硬件无关的命令解析模块。输入是任意长度的字节流（可能半行到达、可能一行太长、可能有空行），输出是“命令名 + 参数 + 状态码”。它要能同时跑在 PC 和目标板上。

需求设计决策原因
逐字节输入cmd_parser_feed(p, ch) 返回 1 表示收到完整一行串口中断就是一个字节一个字节来的
行长上限CMD_LINE_MAX = 64，超长整行丢弃固定内存，不用 malloc；溢出必须显式处理
命令可扩展命令表 + 函数指针新增命令只加一行，不改分发逻辑
错误可区分cmd_status_t 枚举调用者能区分“未知命令 / 参数错 / 越界”
输出可控调用者提供 out 缓冲区与长度不依赖 printf，方便重定向到串口或测试
工程习惯

“解析”和“执行”分开：cmd_parser_feed 只负责切行，cmd_execute 只负责分发。这样解析器可以在没有硬件的情况下单独测试，执行器也可以被脚本直接调用。

13.2接口：cmd.h

#ifndef CMD_H #define CMD_H #include #define CMD_LINE_MAX 64 typedef enum { CMD_OK = 0, CMD_ERR_UNKNOWN, CMD_ERR_ARG, CMD_ERR_RANGE } cmd_status_t; typedef cmd_status_t (*cmd_handler_t)(int argc, char **argv, char *out, size_t out_len); typedef struct { const char *name; cmd_handler_t handler; const char *usage; } cmd_entry_t; typedef struct { char line[CMD_LINE_MAX]; size_t len; int overflow; } cmd_parser_t; void cmd_parser_init(cmd_parser_t *p); int cmd_parser_feed(cmd_parser_t *p, char ch); cmd_status_t cmd_execute(const char *line, char *out, size_t out_len); const char *cmd_status_str(cmd_status_t st); #endif

三个设计点值得注意：

  • cmd_handler_t 的返回值是 cmd_status_t，不是 int。错误码是接口的一部分。
  • out 与 out_len 由调用者提供，模块内部不分配内存，也不直接 printf。
  • cmd_parser_t 把“半行状态”显式保存下来，所以可以跨多次调用继续。

13.3实现：cmd.c

#include "cmd.h" #include #include static cmd_status_t cmd_led(int argc, char **argv, char *out, size_t out_len); static cmd_status_t cmd_read(int argc, char **argv, char *out, size_t out_len); static cmd_status_t cmd_help(int argc, char **argv, char *out, size_t out_len); static const cmd_entry_t g_cmds[] = { { "led", cmd_led, "led on|off" }, { "read", cmd_read, "read <0-3>" }, { "help", cmd_help, "help" }, }; void cmd_parser_init(cmd_parser_t *p) { p->len = 0; p->overflow = 0; } int cmd_parser_feed(cmd_parser_t *p, char ch) { if (ch == '\n' || ch == '\r') { if (p->overflow) { p->len = 0; p->overflow = 0; return -1; /* 上一行太长，已丢弃 */ } p->line[p->len] = '\0'; p->len = 0; return 1; } if (p->len + 1 >= sizeof p->line) { p->overflow = 1; /* 标记溢出，等行结束再丢弃 */ return 0; } p->line[p->len++] = ch; return 0; } static int split(char *s, char **argv, int max) { int argc = 0; while (*s != '\0' && argc < max) { while (*s == ' ' || *s == '\t') { *s++ = '\0'; } if (*s == '\0') { break; } argv[argc++] = s; while (*s != '\0' && *s != ' ' && *s != '\t') { s++; } } return argc; } cmd_status_t cmd_execute(const char *line, char *out, size_t out_len) { char buf[CMD_LINE_MAX]; size_t n = strlen(line); if (n >= sizeof buf) { return CMD_ERR_ARG; } memcpy(buf, line, n + 1); char *argv[6]; int argc = split(buf, argv, 6); if (argc == 0) { return CMD_OK; /* 空行不算错误 */ } for (size_t i = 0; i < sizeof g_cmds / sizeof g_cmds[0]; i++) { if (strcmp(argv[0], g_cmds[i].name) == 0) { return g_cmds[i].handler(argc, argv, out, out_len); } } snprintf(out, out_len, "unknown command: %s", argv[0]); return CMD_ERR_UNKNOWN; } const char *cmd_status_str(cmd_status_t st) { switch (st) { case CMD_OK: return "ok"; case CMD_ERR_UNKNOWN: return "unknown"; case CMD_ERR_ARG: return "bad argument"; case CMD_ERR_RANGE: return "out of range"; default: return "?"; } } static cmd_status_t cmd_led(int argc, char **argv, char *out, size_t out_len) { if (argc != 2) { snprintf(out, out_len, "usage: led on|off"); return CMD_ERR_ARG; } if (strcmp(argv[1], "on") == 0) { snprintf(out, out_len, "LED=1"); return CMD_OK; } if (strcmp(argv[1], "off") == 0) { snprintf(out, out_len, "LED=0"); return CMD_OK; } snprintf(out, out_len, "led: expected on|off"); return CMD_ERR_ARG; } static cmd_status_t cmd_read(int argc, char **argv, char *out, size_t out_len) { if (argc != 2) { snprintf(out, out_len, "usage: read <0-3>"); return CMD_ERR_ARG; } if (argv[1][0] < '0' || argv[1][0] > '3' || argv[1][1] != '\0') { snprintf(out, out_len, "read: channel must be 0..3"); return CMD_ERR_RANGE; } int ch = argv[1][0] - '0'; snprintf(out, out_len, "CH%d=%d", ch, 100 + ch); return CMD_OK; } static cmd_status_t cmd_help(int argc, char **argv, char *out, size_t out_len) { (void)argc; (void)argv; size_t used = 0; used += (size_t)snprintf(out + used, out_len - used, "commands:"); for (size_t i = 0; i < sizeof g_cmds / sizeof g_cmds[0] && used < out_len; i++) { used += (size_t)snprintf(out + used, out_len - used, " %s", g_cmds[i].name); } return CMD_OK; }
易错

split() 直接修改传入的缓冲区，所以 cmd_execute 先把行复制到本地 buf。如果直接改调用者的字符串，遇到字符串字面量就会崩溃——这是“能编译但运行时炸”的典型。

13.4在主机上测试

#include "cmd.h" #include static void run_line(const char *text) { char out[96] = ""; cmd_status_t st = cmd_execute(text, out, sizeof out); printf("> %-16s [%-11s] %s\n", text, cmd_status_str(st), out); } int main(void) { const char *script[] = { "led on", "led off", "led maybe", "read 2", "read 9", "help", "nope", "" }; for (size_t i = 0; i < sizeof script / sizeof script[0]; i++) { run_line(script[i]); } /* 演示逐字节解析：一行被拆成多次喂入 */ cmd_parser_t p; cmd_parser_init(&p); const char *bytes = "led on\n"; for (const char *q = bytes; *q; q++) { if (cmd_parser_feed(&p, *q) == 1) { run_line(p.line); } } return 0; } gcc -std=c11 -Wall -Wextra -Wpedantic -O2 cmd.c main.c -o cmd ./cmd > led on [ok ] LED=1 > led off [ok ] LED=0 > led maybe [bad argument] led: expected on|off > read 2 [ok ] CH2=102 > read 9 [out of range] read: channel must be 0..3 > help [ok ] commands: led read help > nope [unknown ] unknown command: nope > [ok ] > led on [ok ] LED=1

这组输出是实际编译运行的结果。它同时验证了：正常路径、参数错误、越界、未知命令、空行、逐字节喂入。

13.5移植到 MCU：只换输入输出

模块本身不含任何硬件相关代码。移植时只需要把“字节从哪来、结果到哪去”换掉：

#include "cmd.h" #include "ring.h" static cmd_parser_t g_parser; static ring_t g_rx; void app_init(void) { cmd_parser_init(&g_parser); ring_init(&g_rx); uart_enable_rx_interrupt(); } /* 串口中断：只把字节塞进环形缓冲区 */ void USART1_IRQHandler(void) { uint8_t byte = uart_read_byte(); (void)ring_push(&g_rx, byte); } /* 主循环：取出字节、喂给解析器、执行命令 */ void app_poll(void) { uint8_t byte; while (ring_pop(&g_rx, &byte)) { if (cmd_parser_feed(&g_parser, (char)byte) == 1) { char out[96]; cmd_status_t st = cmd_execute(g_parser.line, out, sizeof out); uart_write_str(cmd_status_str(st)); uart_write_str(" "); uart_write_str(out); uart_write_str("\r\n"); } } }

注意 g_parser.line 在 feed 返回 1 之后仍然有效，直到下一次 feed 写入新字符。所以必须在同一轮循环里立刻执行，不能把指针存起来延后处理。

工程习惯

中断里只做 ring_push，解析和执行都在主循环。这样即使一帧命令很长、执行很慢，中断也不会被拖住；环形缓冲区满时丢弃并计数，不会阻塞。

13.6边界与错误处理清单

输入期望行为本实现的处理
空行 / 只有空格忽略，不报错argc == 0 返回 CMD_OK
超长行（≥64 字符）整行丢弃，不截断执行置 overflow，行结束时返回 -1
未知命令返回明确错误CMD_ERR_UNKNOWN + 命令名
参数个数不对返回用法提示CMD_ERR_ARG
参数越界与“参数错”区分CMD_ERR_RANGE
参数过多（>6 个）拒绝，不越界写 argvsplit 的 max 限制
易错

“超长行截断后继续执行”比“整行丢弃”危险得多：用户输入 led onxxxxxxxx... 可能被截成 led on 而意外执行。宁可丢弃并报错。

13.7可以继续扩展的方向

  • 加引号解析，支持带空格的参数（需要更完整的分词器）。
  • 给每条命令加权限等级，区分“只读命令”和“会改配置的命令”。
  • 把 usage 用起来，help <cmd> 输出单条帮助。
  • 加历史记录与上下键（需要终端支持 ANSI 转义）。
  • 把命令表放到 Flash，用 const 减少 RAM 占用。
项目练习 · 第 13 讲
  1. 新增 pwm <0-100> 命令，参数必须是 0–100 的十进制整数，越界返回 CMD_ERR_RANGE。
  2. 让 cmd_parser_feed 支持退格符 \b（删除最后一个字符），并处理行首退格。
  3. 写一个测试函数，输入固定脚本，断言每条命令的状态码与输出。
  4. 解释为什么 cmd_execute 要复制一份输入，而不是直接修改 line。
参考答案与思路

1. 用 strtol 解析并检查 endptr 是否指向结尾，再判断范围；不要用 atoi，它无法区分“0”和“非法输入”。

2. 在 feed 里判断 '\b'：len > 0 时 len--；为 0 时忽略。

3. 用表驱动：{输入, 期望状态, 期望输出}，逐条比较，失败时打印索引与差异。

4. 因为 split 会把空格改成 '\0'；如果调用者传的是字符串字面量或只读内存，写入会崩溃。复制到本地缓冲区是成本最低的防御。

自测：为什么解析器选择“超长行整行丢弃”，而不是“截断后执行”？

安全原则：非法输入应当被拒绝，而不是“尽力猜一个意思”。截断后的前缀可能恰好是一条合法命令，导致用户没有意图的操作被执行。

本讲小结

  • 解析与执行分离，模块才能脱离硬件测试。
  • 命令表 + 函数指针让扩展只加一行。
  • 错误码要能区分未知命令、参数错误和越界。
  • 超长输入整行丢弃，绝不截断执行。
  • 移植时只替换输入输出，核心逻辑不动。
